# C3.4 - Basic Event Recognition

## Collecting Events

Events are collected by looping over `pygame.event.get()` in game loop

*Sample Code*

```py
# I - Import and Initialize
import pygame
pygame.init()

def main():
    '''This function defines the mainline logic for our game.'''
    
    # D - Display configuration
    screen = pygame.display.set_mode((640, 480))
    pygame.display.set_caption("Press mouse and keys - output to Python shell")
    
    # E - Entities
    background = pygame.Surface(screen.get_size())
    background.fill((255, 255, 0))
    
    # A - Action (broken into ALTER steps)
    clock = pygame.time.Clock()
    keepGoing = True
    
    # L - Loop
    while keepGoing:
        # T - Timer to set frame rate
        clock.tick(30)
        
        # E - Event handling
        for event in pygame.event.get():
            if event.type == pygame.QUIT:
                keepGoing = False
            elif event.type == pygame.KEYDOWN:
                keyName = pygame.key.name(event.key)
                print("Key pressed:", keyName)
                if event.key == pygame.K_ESCAPE:
                    keepGoing = False
            elif event.type == pygame.MOUSEBUTTONDOWN:
                print("Mouse down:", pygame.mouse.get_pos())
            elif event.type == pygame.MOUSEBUTTONUP:
                print("Mouse up:", pygame.mouse.get_pos())
        
        # R - Refresh display
        screen.blit(background, (0, 0))
        pygame.display.flip()
    
    pygame.quit()

# Call the main function
main()
```

## Types of Events

- `pygame.QUIT`: closing the game window
- `pygame.KEYDOWN`: pressing key on keyboard
	- get which key by using `event.type == pygame.K_NAME_OF_KEY`
	- get name of key by using `pygame.key.name(event.type)`
	- alt: `pygame.KEYUP` (releasing key)
- `pygame.MOUSEBUTTONDOWN`: pressing mouse button
	- `pygame.mouse.get_pos()` returns (x, y) coords. of mouse position
	- alt: `pygame.MOUSEBUTTONUP` (releasing mouse btn.)
- `pygame.MOUSEMOTION`: movement of the mouse
	- see `MOUSE...DOWN` for mouse-rel. methods

## PyPaint

### PyPaint Skeleton Program

*Code*

```py
# I - Import and Initialize
import pygame
pygame.init()

def main():
    '''This function defines the mainline logic for our game.'''
    
    # D - Display configuration
    screen = pygame.display.set_mode((640, 480))
    pygame.display.set_caption("Draw Lines with the Mouse")
    
    # E - Entities
    background = pygame.Surface(screen.get_size())
    background.fill((255, 255, 255))
    
    # A - Action (broken into ALTER steps)
    clock = pygame.time.Clock()
    keepGoing = True
    lineStart = None  # Initialize variable
    
    # L - Loop
    while keepGoing:
        # T - Timer to set frame rate
        clock.tick(30)
        
        # E - Event handling
        for event in pygame.event.get():
            if event.type == pygame.QUIT:
                keepGoing = False
            elif event.type == pygame.MOUSEBUTTONDOWN:
                lineStart = pygame.mouse.get_pos()
            elif event.type == pygame.MOUSEBUTTONUP and lineStart is not None:
                lineEnd = pygame.mouse.get_pos()
                pygame.draw.line(background, (0, 0, 0), lineStart, lineEnd, 3)
                lineStart = None  # Reset after drawing
        
        # R - Refresh display
        screen.blit(background, (0, 0))
        pygame.display.flip()
    
    pygame.quit()

# Call the main function
main()
```

- When mouse button is pressed, we get starting (x, y) coord.
- When mouse button is released, we get ending (x, y) coord.
- A line is drawn between start and end coords.

### PyPaint with Preview

*Code:*

```py
# I - Import and Initialize
import pygame
pygame.init()

def main():
    '''This function defines the mainline logic for our game.'''
    
    # D - Display configuration
    screen = pygame.display.set_mode((640, 480))
    pygame.display.set_caption("Draw Lines with the Mouse")
    
    # E - Entities
    background = pygame.Surface(screen.get_size())
    background.fill((255, 255, 255))
    
    # A - Action (broken into ALTER steps)
    clock = pygame.time.Clock()
    keepGoing = True
    preview = False
    lineStart = (0, 0)
    lineEnd = (0, 0)
    
    # L - Loop
    while keepGoing:
        # T - Timer to set frame rate
        clock.tick(30)
        
        # E - Event handling
        for event in pygame.event.get():
            if event.type == pygame.QUIT:
                keepGoing = False
            elif event.type == pygame.MOUSEBUTTONDOWN:
                lineStart = pygame.mouse.get_pos()
                preview = True
            elif event.type == pygame.MOUSEBUTTONUP:
                lineEnd = pygame.mouse.get_pos()
                pygame.draw.line(background, (0, 0, 0), lineStart, lineEnd, 3)
                preview = False
            elif event.type == pygame.MOUSEMOTION and preview:
                lineEnd = pygame.mouse.get_pos()
        
        # R - Refresh display
        screen.blit(background, (0, 0))
        if preview:
            pygame.draw.line(screen, (100, 100, 100), lineStart, lineEnd, 1)
        pygame.display.flip()
    
    pygame.quit()

# Call the main function
main()
```

- A boolean `preview` keeps track of whether a preview should be shown or not

- The preview line is hidden when the line has been drawn
- A preview line is drawn when `preview = True`
