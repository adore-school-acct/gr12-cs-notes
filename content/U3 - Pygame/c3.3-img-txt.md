# C3.3 - Images and Text

## Loading Image

- `pygame.image.load(path_to_img)` loads an image as a surface from `path_to_img`
- An image must be blitted separately
- `convert()` method converts any image format into pygame format
- `image_var.set_colorkey(color)` makes the parts where the color is `color` transparent
	- i.e. `color = "white"` will make all white portions of image transparent

```py
# I - Import and Initialize
import pygame
pygame.init()

def main():
     '''This function defines the 'mainline logic' for our game.'''
     # D - Display configuration
     screen = pygame.display.set_mode((640, 480))
     pygame.display.set_caption("Load an image")
	 
     # E - Entities
     background = pygame.Surface(screen.get_size())
     background.fill((0, 0, 255))
     flower = pygame.image.load("flower.png") #!
	 
     # A - Action (broken into ALTER steps)
     # A - Assign values to key variables
     clock = pygame.time.Clock()
     keepGoing = True

     # L - Loop
     while keepGoing:
          # T - Timer to set frame rate
          clock.tick(30)
		  
          # E - Event handling
          for event in pygame.event.get():
               if event.type == pygame.QUIT:
                    keepGoing = False

          # R - Refresh display
          screen.blit(background, (0, 0))
          screen.blit(flower, (60, 60)) #!
          pygame.display.flip()

# Call the main function
main()
```

## Saving Image

`pygame.image.save(path)` saves an image of a surface to a file

Supported formats: png, gif, bitmap (bmp), jpeg

```py
# I - Import and Initialize
import pygame
pygame.init()

def drawLines(background):
    '''Draw an interesting pattern of lines on a given surface.'''
    STEPS = 15
    width = background.get_width()
    height = background.get_height()
    widthStepSize = width / STEPS
    heightStepSize = height / STEPS
    
    for i in range(STEPS + 1):
        pygame.draw.line(background, (0, 0, 0), (0, i * heightStepSize),
                         (i * widthStepSize, height))
        pygame.draw.line(background, (0, 0, 0), (i * widthStepSize, 0),
                         (width, i * heightStepSize))
        pygame.draw.line(background, (0, 0, 0), (width, i * heightStepSize),
                         (width - (i * widthStepSize), height))
        pygame.draw.line(background, (0, 0, 0), (i * widthStepSize, 0),
                         (0, height - (i * heightStepSize)))

def main():
    '''This function defines the main logic for our program.'''
    # D - Display configuration
    # (No display window will open in this example.)
    
    # E - Entities
    background = pygame.Surface((640, 480))
    background.fill((255, 255, 255))
    drawLines(background)
    
    # Save the background surface as a PNG file
    pygame.image.save(background, "lines.png")

# Call the main function
main()
```

### Display Text

- Create a `pygame.font` object to store font characteristics
	- Use system font to use built-in fonts
	- Use custom font (TTF, TrueType Font)
- `pygame.font.SysFont(font, size)` imports a system font into the game
	- `font`: name of font
	- `size`: size of font (px)
- `pygame.font.Font(path_to_ttf, size)` imports a custom font into game
	- `path_to_ttf`: path to TTF file where font is stored
	- `size`: size of font (px)
- Text is made using `font_var.render(text)`
	- `font_var`: the font object
	- `text`: text to be displayed
- Text must also be blitted on the screen

```py
# I - Import and Initialize
import pygame
pygame.init()

def main():
    '''This function defines the mainline logic for our game.'''
    
    # D - Display configuration
    screen = pygame.display.set_mode((640, 480))
    pygame.display.set_caption("Display Some Text")
    
    # E - Entities
    background = pygame.Surface(screen.get_size())
    background.fill((0, 0, 0))
    
    mySystemFont = pygame.font.SysFont("Arial", 60)
    label1 = mySystemFont.render("Hello World!", True, (255, 255, 0))
    
    # Make sure you have "Air Millhouse Italic.ttf" in the same folder
    # or replace with another font file you have
    try:
        myCustomFont = pygame.font.Font("Air Millhouse Italic.ttf", 50)
    except FileNotFoundError:
        myCustomFont = pygame.font.SysFont("Comic Sans MS", 50)
    
    label2 = myCustomFont.render("Howdy Partner!", True, (255, 255, 0))
    
    # A - Action (broken into ALTER steps)
    clock = pygame.time.Clock()
    keepGoing = True
    
    # L - Loop
    while keepGoing:
        # T - Timer to set frame rate
        clock.tick(30)
        
        # E - Event handling
        for event in pygame.event.get():
            if event.type == pygame.QUIT:
                keepGoing = False
        
        # R - Refresh display
        screen.blit(background, (0, 0))
        screen.blit(label1, (100, 100))
        screen.blit(label2, (100, 200))
        pygame.display.flip()

# Call the main function
main()
```
